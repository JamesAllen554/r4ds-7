prices <- read_parquet("data/daily_prices.parquet")
prices |>
  filter(coin_id == "btc") |>
  ggplot(aes(x = as.Date(date_raw), y = price_usd)) +
  geom_line() +
  labs(
    x = "Date",
    y = "Price (USD)",
    title = "Bitcoin daily price"
  )
prices |>
  inner_join(read_parquet("data/coin_metadata.parquet"), by = "coin_id") |>
  inner_join(read_parquet("data/categories.parquet"), by = "category_id") |>
  group_by(date_raw, category_name) |>
  summarise(category_market_cap = sum(market_cap_usd, na.rm = TRUE), .groups = "drop") |>
  group_by(date_raw) |>
  mutate(category_share = category_market_cap / sum(category_market_cap)) |>
  ungroup() -> category_shares

category_order <- category_shares |>
  group_by(category_name) |>
  summarise(typical_share = median(category_share), .groups = "drop") |>
  arrange(desc(typical_share)) |>
  pull(category_name)
